DECLARE
  v_owner_id       uuid;
  v_blueprint_id   bigint;
  v_unit_name      text;
  v_shipyard_level integer;
  v_owned          integer;
  v_validation     jsonb;
BEGIN
  -- Solo naves o defensas
  IF NEW.item_type NOT IN ('ship','defense') THEN
    RETURN NEW;
  END IF;
  IF NEW.item_type = 'ship'    AND NEW.ship_type_id    IS NULL THEN RETURN NEW; END IF;
  IF NEW.item_type = 'defense' AND NEW.defense_type_id IS NULL THEN RETURN NEW; END IF;

  SELECT p.user_id INTO v_owner_id FROM planets p WHERE p.id = NEW.planet_id;
  IF v_owner_id IS NULL THEN
    RETURN NEW;
  END IF;

  -- Localizar plano (nave o defensa)
  IF NEW.item_type = 'ship' THEN
    SELECT sb.id, st.name INTO v_blueprint_id, v_unit_name
    FROM ship_blueprints sb
    JOIN ship_types st ON st.id = sb.ship_type_id
    WHERE sb.ship_type_id = NEW.ship_type_id AND sb.is_active = true
    LIMIT 1;
  ELSE
    SELECT sb.id, dt.name INTO v_blueprint_id, v_unit_name
    FROM ship_blueprints sb
    JOIN defense_types dt ON dt.id = sb.defense_type_id
    WHERE sb.defense_type_id = NEW.defense_type_id AND sb.is_active = true
    LIMIT 1;
  END IF;

  -- Sin plano definido: sin restriccion extra (via normal ya validada)
  IF v_blueprint_id IS NULL THEN
    RETURN NEW;
  END IF;

  SELECT COUNT(*)::integer INTO v_owned
  FROM user_ship_blueprints usb
  WHERE usb.user_id = v_owner_id AND usb.blueprint_id = v_blueprint_id;

  -- CONSUMIBLE: si posee planos suficientes para toda la tanda,
  -- exige Hangar Nv1, consume NEW.quantity planos (los mas antiguos) y listo.
  IF v_owned >= NEW.quantity THEN
    SELECT COALESCE(pb.level, 0) INTO v_shipyard_level
    FROM planet_buildings pb
    JOIN building_types bt ON bt.id = pb.building_type_id
    WHERE pb.planet_id = NEW.planet_id AND bt.code = 'shipyard'
    LIMIT 1;

    IF COALESCE(v_shipyard_level, 0) < 1 THEN
      RAISE EXCEPTION 'Necesitas Hangar nivel 1 para construir con plano.'
        USING ERRCODE = 'P0001';
    END IF;

    DELETE FROM user_ship_blueprints
    WHERE ctid IN (
      SELECT usb.ctid FROM user_ship_blueprints usb
      WHERE usb.user_id = v_owner_id AND usb.blueprint_id = v_blueprint_id
      ORDER BY usb.acquired_at ASC
      LIMIT NEW.quantity
    );
    NEW.blueprints_used := NEW.quantity;
    RETURN NEW;
  END IF;

  -- Sin planos suficientes: requisitos completos (via normal)
  IF NEW.item_type = 'ship' THEN
    v_validation := validate_ship_requirements(NEW.planet_id, NEW.ship_type_id);
  ELSE
    v_validation := validate_defense_requirements(NEW.planet_id, NEW.defense_type_id);
  END IF;

  IF NOT COALESCE((v_validation->>'valid')::boolean, false) THEN
    RAISE EXCEPTION 'No cumples los requisitos para construir %. Compra su plano o cumple los requisitos.',
      COALESCE(v_unit_name, 'esto')
      USING ERRCODE = 'P0001';
  END IF;

  NEW.blueprints_used := 0;
  RETURN NEW;
END;

CREATE OR REPLACE FUNCTION public.user_can_build_ship(p_user_id uuid, p_ship_type_id integer)
 RETURNS boolean
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
DECLARE
  v_requirement record;
  v_user_tech_level integer;
  v_user_building_level integer;
BEGIN
  FOR v_requirement IN
    SELECT sr.required_tech_id, sr.required_tech_level,
           sr.required_building_id, sr.required_building_level
    FROM ship_requirements sr
    WHERE sr.ship_type_id = p_ship_type_id
  LOOP
    IF v_requirement.required_tech_id IS NOT NULL AND COALESCE(v_requirement.required_tech_level, 0) > 0 THEN
      SELECT COALESCE(ut.level, 0) INTO v_user_tech_level
      FROM user_technologies ut
      WHERE ut.user_id = p_user_id
        AND ut.technology_type_id = v_requirement.required_tech_id;

      IF COALESCE(v_user_tech_level, 0) < v_requirement.required_tech_level THEN
        RETURN false;
      END IF;
    END IF;

    IF v_requirement.required_building_id IS NOT NULL AND COALESCE(v_requirement.required_building_level, 0) > 0 THEN
      SELECT COALESCE(MAX(pb.level), 0) INTO v_user_building_level
      FROM planet_buildings pb
      JOIN planets p ON p.id = pb.planet_id
      WHERE p.user_id = p_user_id
        AND pb.building_type_id = v_requirement.required_building_id;

      IF COALESCE(v_user_building_level, 0) < v_requirement.required_building_level THEN
        RETURN false;
      END IF;
    END IF;
  END LOOP;

  RETURN true;
END;

DECLARE
  v_requirement RECORD;
  v_current_level integer;
  v_missing_reqs jsonb := '[]'::jsonb;
  v_planet RECORD;
BEGIN
  SELECT * INTO v_planet FROM planets WHERE id = p_planet_id;
  IF NOT FOUND THEN
    RETURN jsonb_build_object('valid', false, 'error', 'Planet not found');
  END IF;

  FOR v_requirement IN
    SELECT sr.*, 
           bt.name as building_name, 
           tt.name as tech_name
    FROM ship_requirements sr
    LEFT JOIN building_types bt ON bt.id = sr.required_building_id
    LEFT JOIN technology_types tt ON tt.id = sr.required_tech_id
    WHERE sr.ship_type_id = p_ship_type_id
  LOOP
    IF v_requirement.required_building_id IS NOT NULL THEN
      SELECT COALESCE(pb.level, 0) INTO v_current_level
      FROM planet_buildings pb
      WHERE pb.planet_id = p_planet_id
        AND pb.building_type_id = v_requirement.required_building_id;
      
      IF v_current_level IS NULL THEN v_current_level := 0; END IF;
      
      IF v_current_level < v_requirement.required_building_level THEN
        v_missing_reqs := v_missing_reqs || jsonb_build_object(
          'type', 'building',
          'name', v_requirement.building_name,
          'required', v_requirement.required_building_level,
          'current', v_current_level
        );
      END IF;
    END IF;

    IF v_requirement.required_tech_id IS NOT NULL THEN
      SELECT COALESCE(ut.level, 0) INTO v_current_level
      FROM user_technologies ut
      WHERE ut.user_id = v_planet.user_id
        AND ut.technology_type_id = v_requirement.required_tech_id;
      
      IF v_current_level IS NULL THEN v_current_level := 0; END IF;
      
      IF v_current_level < v_requirement.required_tech_level THEN
        v_missing_reqs := v_missing_reqs || jsonb_build_object(
          'type', 'technology',
          'name', v_requirement.tech_name,
          'required', v_requirement.required_tech_level,
          'current', v_current_level
        );
      END IF;
    END IF;
  END LOOP;

  IF jsonb_array_length(v_missing_reqs) > 0 THEN
    RETURN jsonb_build_object('valid', false, 'missing', v_missing_reqs);
  END IF;

  RETURN jsonb_build_object('valid', true);
END;

DECLARE

    RETURN;
  END IF;

  UPDATE public.battleship_entries
     SET payout = CASE WHEN user_id = p_winner THEN m.pot ELSE 0 END
   WHERE match_id = p_match_id;
  UPDATE public.battleship_matches
     SET status = 'finished', winner = p_winner, end_reason = p_reason, payout = m.pot,

   WHERE id = p_match_id;

  IF NOT m.loser_fee THEN
    -- Partidas antiguas: la comision ya se cobro al unirse
    UPDATE public.battleship_config SET total_paid = total_paid + m.pot, updated_at = now() WHERE id = 1;
    RETURN;
  END IF;

  v_loser   := CASE WHEN p_winner = m.player1 THEN m.player2 ELSE m.player1 END;
  v_fee     := m.stake * 2 - m.pot;                -- 20% de la apuesta del perdedor
  v_pool    := round(m.stake * c_pool_rate, 8);    -- 5%  pools normales
  v_daily   := round(m.stake * c_daily_rate, 8);   -- 5%  pool diario Batalla de Naves

  SELECT p.referred_by INTO v_referrer FROM public.profiles p WHERE p.id = v_loser;
  IF v_referrer IS NOT NULL AND v_referrer <> v_loser
     AND EXISTS (SELECT 1 FROM public.profiles WHERE id = v_referrer) THEN
    v_ref_share := round(v_company * c_ref_rate, 8);
  ELSE
    v_referrer := NULL;
    v_ref_share := 0;
  END IF;
  v_admin_share := v_company - v_ref_share;

  SELECT id INTO v_admin_id FROM public.profiles WHERE is_admin = true ORDER BY created_at ASC LIMIT 1;
