CREATE OR REPLACE FUNCTION public.admin_get_planet_inventory(p_planet_id uuid)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
DECLARE
  v_is_admin boolean;
  v_planet RECORD;
  v_ships jsonb;
  v_buildings jsonb;
  v_defenses jsonb;
BEGIN
  SELECT is_admin INTO v_is_admin FROM profiles WHERE id = auth.uid();
  IF v_is_admin IS NOT TRUE THEN
    RAISE EXCEPTION 'Access denied: admin only';
  END IF;

  SELECT p.id, p.name, p.user_id, p.galaxy, p.solar_system, p."position",
         p.planet_type, p.is_main_planet, pr.username
    INTO v_planet
  FROM planets p
  LEFT JOIN profiles pr ON pr.id = p.user_id
  WHERE p.id = p_planet_id;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Planet not found';
  END IF;

  -- Naves: TODOS los ship_types con la cantidad actual (0 si no existe)
  SELECT COALESCE(jsonb_agg(jsonb_build_object(
    'ship_type_id', st.id,
    'code',         st.code,
    'name',         st.name,
    'quantity',     COALESCE(ps.quantity, 0)
  ) ORDER BY st.id), '[]'::jsonb)
  INTO v_ships
  FROM ship_types st
  LEFT JOIN planet_ships ps ON ps.ship_type_id = st.id AND ps.planet_id = p_planet_id;

  -- Edificios: TODOS los building_types con su nivel actual (0 si no existe)
  SELECT COALESCE(jsonb_agg(jsonb_build_object(
    'building_type_id', bt.id,
    'code',             bt.code,
    'name',             bt.name,
    'level',            COALESCE(pb.level, 0),
    'is_upgrading',     COALESCE(pb.is_upgrading, false)
  ) ORDER BY bt.id), '[]'::jsonb)
  INTO v_buildings
  FROM building_types bt
  LEFT JOIN planet_buildings pb ON pb.building_type_id = bt.id AND pb.planet_id = p_planet_id;

  -- Defensas: TODOS los defense_types con la cantidad actual (0 si no existe)
  SELECT COALESCE(jsonb_agg(jsonb_build_object(
    'defense_type_id', dt.id,
    'code',            dt.code,
    'name',            dt.name,
    'quantity',        COALESCE(pd.quantity, 0),
    'max_per_planet',  dt.max_per_planet
  ) ORDER BY dt.id), '[]'::jsonb)
  INTO v_defenses
  FROM defense_types dt
  LEFT JOIN planet_defenses pd ON pd.defense_type_id = dt.id AND pd.planet_id = p_planet_id;

  RETURN jsonb_build_object(
    'planet', jsonb_build_object(
      'id',             v_planet.id,
      'name',           v_planet.name,
      'user_id',        v_planet.user_id,
      'username',       v_planet.username,
      'coords',         format('[%s:%s:%s]', v_planet.galaxy, v_planet.solar_system, v_planet."position"),
      'planet_type',    v_planet.planet_type,
      'is_main_planet', v_planet.is_main_planet
    ),
    'ships',     v_ships,
    'buildings', v_buildings,
    'defenses',  v_defenses
  );
END;
$function$

CREATE OR REPLACE FUNCTION public.admin_set_planet_ship(p_planet_id uuid, p_ship_type_id integer, p_quantity integer)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
DECLARE
  v_is_admin boolean;
  v_old_qty integer;
  v_new_qty integer;
  v_ship_name text;
  v_planet_exists boolean;
BEGIN
  SELECT is_admin INTO v_is_admin FROM profiles WHERE id = auth.uid();
  IF v_is_admin IS NOT TRUE THEN
    RAISE EXCEPTION 'Access denied: admin only';
  END IF;

  IF p_quantity < 0 THEN
    RAISE EXCEPTION 'Quantity must be >= 0';
  END IF;

  SELECT EXISTS(SELECT 1 FROM planets WHERE id = p_planet_id) INTO v_planet_exists;
  IF NOT v_planet_exists THEN
    RAISE EXCEPTION 'Planet not found';
  END IF;

  SELECT name INTO v_ship_name FROM ship_types WHERE id = p_ship_type_id;
  IF v_ship_name IS NULL THEN
    RAISE EXCEPTION 'Ship type not found: %', p_ship_type_id;
  END IF;

  SELECT COALESCE(quantity, 0) INTO v_old_qty
  FROM planet_ships
  WHERE planet_id = p_planet_id AND ship_type_id = p_ship_type_id;
  v_old_qty := COALESCE(v_old_qty, 0);

  IF p_quantity = 0 THEN
    DELETE FROM planet_ships
    WHERE planet_id = p_planet_id AND ship_type_id = p_ship_type_id;
    v_new_qty := 0;
  ELSE
    INSERT INTO planet_ships (planet_id, ship_type_id, quantity)
    VALUES (p_planet_id, p_ship_type_id, p_quantity)
    ON CONFLICT (planet_id, ship_type_id)
      DO UPDATE SET quantity = EXCLUDED.quantity;
    v_new_qty := p_quantity;
  END IF;

  RETURN jsonb_build_object(
    'success',   true,
    'planet_id', p_planet_id,
    'ship_type_id', p_ship_type_id,
    'ship_name', v_ship_name,
    'old_quantity', v_old_qty,
    'new_quantity', v_new_qty
  );
END;
$function$

CREATE OR REPLACE FUNCTION public.occupation_abandon(p_occupation_id uuid)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
DECLARE
  v_uid uuid := auth.uid();
  o public.occupations%ROWTYPE;
BEGIN
  IF v_uid IS NULL THEN
    RETURN jsonb_build_object('success', false, 'error', 'No autenticado');
  END IF;
  SELECT * INTO o FROM public.occupations
  WHERE id = p_occupation_id AND occupier_id = v_uid AND status = 'active' FOR UPDATE;
  IF NOT FOUND THEN
    RETURN jsonb_build_object('success', false, 'error', 'Ocupación no encontrada');
  END IF;

  -- Las unidades ya terminadas se suman a la guarnicion antes de partir
  PERFORM public._occ_process_builds(o.id);

  UPDATE public.fleets
  SET is_returning = true,
      arrival_time = now(),
      return_time = now() + GREATEST(interval '10 seconds', arrival_time - departure_time)
  WHERE id = o.garrison_fleet_id AND NOT is_returning AND NOT is_completed;

  -- Si la guarnicion ya no existia, cerrar igualmente
  PERFORM public._occ_end(o.id, 'abandoned');

  RETURN jsonb_build_object('success', true, 'message', 'Has abandonado la ocupación. Tu guarnición regresa a casa.');
END
$function$
