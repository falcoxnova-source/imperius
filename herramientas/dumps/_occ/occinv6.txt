CREATE OR REPLACE FUNCTION public.fleet_calc_fuel(p_fleet_ships jsonb, p_distance integer, p_speed_percent integer)
 RETURNS bigint
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
DECLARE
  v_ship record;
  v_ship_type record;
  v_total_fuel bigint := 0;
  v_ship_fuel numeric;
  v_distance_factor numeric;
  v_speed_factor numeric;
BEGIN
  v_distance_factor := p_distance::numeric / 35000;
  v_speed_factor := (p_speed_percent::numeric / 100) + 1;
  
  FOR v_ship IN SELECT * FROM jsonb_to_recordset(p_fleet_ships) AS x(ship_type_id int, quantity int)
  LOOP
    IF v_ship.quantity > 0 THEN
      SELECT * INTO v_ship_type FROM ship_types WHERE id = v_ship.ship_type_id;
      
      IF v_ship_type IS NOT NULL THEN
        v_ship_fuel := v_ship_type.fuel_consumption * v_ship.quantity * v_distance_factor * v_speed_factor;
        v_total_fuel := v_total_fuel + CEIL(v_ship_fuel);
      END IF;
    END IF;
  END LOOP;
  
  RETURN v_total_fuel;
END;
$function$

CREATE OR REPLACE FUNCTION public.fleet_calc_slowest_speed(p_fleet_ships jsonb, p_user_id uuid)
 RETURNS integer
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
DECLARE
  v_ship record;
  v_ship_type record;
  v_tech_combustion int := 0;
  v_tech_impulse int := 0;
  v_tech_hyperspace int := 0;
  v_speed int;
  v_slowest int := 999999999;
BEGIN
  SELECT COALESCE(
    (SELECT level FROM user_technologies ut 
     JOIN technology_types tt ON ut.technology_type_id = tt.id 
     WHERE ut.user_id = p_user_id AND tt.code = 'combustion_drive'), 0
  ) INTO v_tech_combustion;
  
  SELECT COALESCE(
    (SELECT level FROM user_technologies ut 
     JOIN technology_types tt ON ut.technology_type_id = tt.id 
     WHERE ut.user_id = p_user_id AND tt.code = 'impulse_drive'), 0
  ) INTO v_tech_impulse;
  
  SELECT COALESCE(
    (SELECT level FROM user_technologies ut 
     JOIN technology_types tt ON ut.technology_type_id = tt.id 
     WHERE ut.user_id = p_user_id AND tt.code = 'hyperspace_drive'), 0
  ) INTO v_tech_hyperspace;

  FOR v_ship IN SELECT * FROM jsonb_to_recordset(p_fleet_ships) AS x(ship_type_id int, quantity int)
  LOOP
    IF v_ship.quantity > 0 THEN
      SELECT * INTO v_ship_type FROM ship_types WHERE id = v_ship.ship_type_id;
      
      IF v_ship_type.base_speed > 0 THEN
        v_speed := fleet_calc_ship_speed(
          v_ship_type.code,
          v_ship_type.base_speed,
          v_tech_combustion,
          v_tech_impulse,
          v_tech_hyperspace
        );
        
        IF v_speed < v_slowest THEN
          v_slowest := v_speed;
        END IF;
      END IF;
    END IF;
  END LOOP;
  
  IF v_slowest = 999999999 THEN
    RETURN 0;
  END IF;
  
  RETURN v_slowest;
END;
$function$

CREATE OR REPLACE FUNCTION public.fleet_calc_flight_time(p_distance integer, p_slowest_speed integer, p_speed_percent integer, p_admiral_bonus numeric DEFAULT 0)
 RETURNS integer
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
DECLARE
  v_fleet_speed numeric;
  v_effective_speed numeric;
  v_flight_time int;
BEGIN
  IF p_slowest_speed <= 0 THEN
    RETURN 0;
  END IF;
  
  v_fleet_speed := get_fleet_speed();
  v_effective_speed := p_slowest_speed * (p_speed_percent::numeric / 100);
  
  v_flight_time := GREATEST(
    10,
    ROUND(
      (35000::numeric / (p_speed_percent::numeric / 100) * 
       SQRT(p_distance::numeric * 10 / v_effective_speed) + 10) / v_fleet_speed
    )
  );
  
  -- Apply admiral bonus: reduce flight time by the bonus percentage
  -- e.g. p_admiral_bonus = 0.2 means 20% faster ??? divide by 1.2
  IF p_admiral_bonus > 0 THEN
    v_flight_time := GREATEST(10, ROUND(v_flight_time::numeric / (1 + p_admiral_bonus)));
  END IF;
  
  RETURN v_flight_time;
END;
$function$

CREATE OR REPLACE FUNCTION public.fleet_calc_distance(p_origin_galaxy integer, p_origin_system integer, p_origin_position integer, p_target_galaxy integer, p_target_system integer, p_target_position integer)
 RETURNS integer
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
BEGIN
  IF p_origin_galaxy != p_target_galaxy THEN
    RETURN ABS(p_origin_galaxy - p_target_galaxy) * 20000;
  END IF;
  
  IF p_origin_system != p_target_system THEN
    RETURN ABS(p_origin_system - p_target_system) * 95 + 2700;
  END IF;
  
  IF p_origin_position != p_target_position THEN
    RETURN ABS(p_origin_position - p_target_position) * 5 + 1000;
  END IF;
  
  RETURN 5;
END;
$function$

CREATE OR REPLACE FUNCTION public.apply_vip_time_bonus(p_user_id uuid, p_seconds integer)
 RETURNS integer
 LANGUAGE plpgsql
 STABLE SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
DECLARE
  v_mult numeric;
BEGIN
  IF p_seconds IS NULL OR p_user_id IS NULL THEN
    RETURN p_seconds;
  END IF;
  IF NOT public.is_user_plus(p_user_id) THEN
    RETURN p_seconds;
  END IF;
  v_mult := public.get_vip_time_speed_multiplier();
  IF v_mult IS NULL OR v_mult <= 1 THEN
    RETURN p_seconds;
  END IF;
  RETURN GREATEST(1, FLOOR(p_seconds::numeric / v_mult))::integer;
END;
$function$

CREATE OR REPLACE FUNCTION public.enforce_shield_dome_per_account()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
DECLARE
  v_total_acct integer;
  v_user_id uuid;
  v_def_name text;
BEGIN
  IF NEW.defense_type_id NOT IN (7, 8) THEN
    RETURN NEW;
  END IF;

  IF NEW.quantity <= 0 THEN
    RETURN NEW;
  END IF;

  SELECT user_id INTO v_user_id FROM planets WHERE id = NEW.planet_id;

  SELECT COALESCE(SUM(pd.quantity), 0) INTO v_total_acct
  FROM planet_defenses pd
  JOIN planets pl ON pl.id = pd.planet_id
  WHERE pl.user_id = v_user_id
    AND pd.defense_type_id = NEW.defense_type_id
    AND pd.id IS DISTINCT FROM NEW.id;

  IF v_total_acct + NEW.quantity > 1 THEN
    SELECT name INTO v_def_name FROM defense_types WHERE id = NEW.defense_type_id;
    RAISE EXCEPTION 'L??mite: solo 1 % por cuenta permitido. Ya tienes %.', v_def_name, v_total_acct;
  END IF;

  RETURN NEW;
END;
$function$
