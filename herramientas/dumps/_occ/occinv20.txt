== sync_planet_resources

DECLARE
  v_calculated jsonb;
  v_user_id uuid;
BEGIN
  -- Verify ownership
  SELECT user_id INTO v_user_id FROM planets WHERE id = p_planet_id;
  IF v_user_id IS NULL OR v_user_id != auth.uid() THEN
    RETURN jsonb_build_object('error', 'Unauthorized');
  END IF;
  
  -- Calculate current resources
  v_calculated := calculate_planet_resources(p_planet_id);
  
  IF v_calculated ? 'error' THEN
    RETURN v_calculated;
  END IF;
  
  -- Update the database
  UPDATE planet_resources
  SET 
    metal = (v_calculated->>'metal')::numeric,
    crystal = (v_calculated->>'crystal')::numeric,
    deuterium = (v_calculated->>'deuterium')::numeric,
    metal_per_hour = (v_calculated->>'metal_per_hour')::numeric,
    crystal_per_hour = (v_calculated->>'crystal_per_hour')::numeric,
    deuterium_per_hour = (v_calculated->>'deuterium_per_hour')::numeric,
    metal_storage = (v_calculated->>'metal_storage')::numeric,
    crystal_storage = (v_calculated->>'crystal_storage')::numeric,
    deuterium_storage = (v_calculated->>'deuterium_storage')::numeric,
    energy_used = (v_calculated->>'energy_used')::numeric,
    energy_max = (v_calculated->>'energy_max')::numeric,
    last_update = now()
  WHERE planet_id = p_planet_id;
  
  -- Return updated values with new timestamp
  RETURN jsonb_build_object(
    'metal', (v_calculated->>'metal')::numeric,
    'crystal', (v_calculated->>'crystal')::numeric,
    'deuterium', (v_calculated->>'deuterium')::numeric,
    'metal_per_hour', (v_calculated->>'metal_per_hour')::numeric,
    'crystal_per_hour', (v_calculated->>'crystal_per_hour')::numeric,
    'deuterium_per_hour', (v_calculated->>'deuterium_per_hour')::numeric,
    'metal_storage', (v_calculated->>'metal_storage')::numeric,
    'crystal_storage', (v_calculated->>'crystal_storage')::numeric,
    'deuterium_storage', (v_calculated->>'deuterium_storage')::numeric,
    'energy_used', (v_calculated->>'energy_used')::numeric,
    'energy_max', (v_calculated->>'energy_max')::numeric,
    'last_update', now()
  );
END;

== recall_deployed_fleet

DECLARE
  v_user_id uuid;
  v_fleet RECORD;
  v_now timestamptz;
  v_original_flight_seconds numeric;
  v_return_time timestamptz;
BEGIN
  v_user_id := auth.uid();

  IF v_user_id IS NULL THEN
    RETURN jsonb_build_object('success', false, 'error', 'No autenticado');
  END IF;

  SELECT * INTO v_fleet
  FROM fleets
  WHERE id = p_fleet_id
  FOR UPDATE;

  IF v_fleet IS NULL THEN
    RETURN jsonb_build_object('success', false, 'error', 'Flota no encontrada');
  END IF;

  IF v_fleet.user_id != v_user_id THEN
    RETURN jsonb_build_object('success', false, 'error', 'No puedes retirar flotas de otros jugadores');
  END IF;

  IF v_fleet.is_completed THEN
    RETURN jsonb_build_object('success', false, 'error', 'La flota ya completo su mision');
  END IF;

  IF v_fleet.is_returning THEN
    RETURN jsonb_build_object('success', false, 'error', 'La flota ya esta regresando');
  END IF;

  IF v_fleet.mission_type != 'deploy' THEN
    RETURN jsonb_build_object('success', false, 'error', 'Solo se pueden retirar flotas en mision de estacionamiento');
  END IF;

  v_now := now();

  -- Fleet must have already arrived (i.e. it's stationed, not in transit)
  IF v_fleet.arrival_time > v_now THEN
    RETURN jsonb_build_object('success', false, 'error', 'La flota aun no ha llegado. Usa cancelar flota en su lugar');
  END IF;

  -- Calculate return time = same duration as original flight
  v_original_flight_seconds := EXTRACT(EPOCH FROM (v_fleet.arrival_time - v_fleet.departure_time));
  IF v_original_flight_seconds < 10 THEN
    v_original_flight_seconds := 10;
  END IF;

  v_return_time := v_now + (v_original_flight_seconds || ' seconds')::interval;

  UPDATE fleets
  SET is_returning = true,
      arrival_time = v_now,
      return_time = v_return_time
  WHERE id = p_fleet_id;

  RETURN jsonb_build_object(
    'success', true,
    'message', 'Flota retirada. Regresara en ' || CEIL(v_original_flight_seconds) || ' segundos',
    'return_seconds', CEIL(v_original_flight_seconds)
  );

EXCEPTION
  WHEN OTHERS THEN
    RETURN jsonb_build_object('success', false, 'error', 'Error al retirar flota: ' || SQLERRM);
END;

== get_live_tung_price_usd

DECLARE
  v_value   text;
  v_updated timestamptz;
  v_price   numeric;
BEGIN
  SELECT value, updated_at INTO v_value, v_updated
  FROM game_settings
  WHERE key = 'tung_price_usd';

  IF v_value IS NULL THEN
    RETURN NULL;
  END IF;

  -- Ventana de tolerancia: 12 horas.
  -- Mejor un precio de hace unas horas que dejar el juego sin precios.
  IF v_updated IS NULL OR v_updated < now() - interval '12 hours' THEN
    RETURN NULL;
  END IF;

  BEGIN
    v_price := v_value::numeric;
  EXCEPTION WHEN others THEN
    RETURN NULL;
  END;

  IF v_price IS NULL OR v_price <= 0 THEN
    RETURN NULL;
  END IF;

  RETURN v_price;
END;

== protect_game_table_full

DECLARE
  v_cu text := current_user;
BEGIN
  -- Bypass para roles no expuestos al cliente (postgres, service_role, etc.)
  IF v_cu NOT IN ('authenticated', 'anon') THEN
    RETURN NEW;
  END IF;

  -- Bypass de emergencia compatible con el patr??n existente.
  IF current_setting('fxu.allow_update_9x7k2m', true) = 'true' THEN
    RETURN NEW;
  END IF;

  RAISE EXCEPTION
    'SECURITY_VIOLATION: Direct UPDATE on % from client is not allowed. Use the proper RPC. Code: GAME001',
    TG_TABLE_NAME
    USING ERRCODE = 'insufficient_privilege';
END;
