== contribute_to_jackpot

DECLARE
  v_expedition_amount numeric;
  v_monthly_amount numeric;
  v_plus_amount numeric;
  v_destruction_amount numeric;
  v_expedition_percent int;
  v_monthly_percent int;
  v_plus_percent int;
  v_destruction_percent int;
  v_new_total numeric;
BEGIN
  IF p_amount <= 0 THEN
    RETURN 0;
  END IF;

  SELECT expedition_percent, monthly_percent, plus_percent, destruction_percent
  INTO v_expedition_percent, v_monthly_percent, v_plus_percent, v_destruction_percent
  FROM jackpot_pools WHERE id = 1;

  IF v_expedition_percent IS NULL THEN
    v_expedition_percent := 10;
    v_monthly_percent := 10;
    v_plus_percent := 50;
    v_destruction_percent := 30;
  END IF;
  IF v_destruction_percent IS NULL THEN
    v_destruction_percent := 40;
  END IF;

  v_expedition_amount := FLOOR(p_amount * v_expedition_percent / 100);
  v_monthly_amount := FLOOR(p_amount * v_monthly_percent / 100);
  v_plus_amount := FLOOR(p_amount * v_plus_percent / 100);
  v_destruction_amount := FLOOR(p_amount * v_destruction_percent / 100);

  UPDATE jackpot_pools
  SET 
    expedition_balance = expedition_balance + v_expedition_amount,
    expedition_total = expedition_total + v_expedition_amount,
    monthly_balance = monthly_balance + v_monthly_amount,
    monthly_total = monthly_total + v_monthly_amount,
    plus_balance = plus_balance + v_plus_amount,
    plus_total = plus_total + v_plus_amount,
    destruction_balance = destruction_balance + v_destruction_amount,
    destruction_total = destruction_total + v_destruction_amount,
    updated_at = now()
  WHERE id = 1;

  UPDATE jackpot
  SET balance = balance + p_amount, total_contributed = total_contributed + p_amount, updated_at = now()
  WHERE id = 1;

  SELECT expedition_balance + monthly_balance + plus_balance + destruction_balance 
  INTO v_new_total
  FROM jackpot_pools WHERE id = 1;

  INSERT INTO jackpot_history (user_id, type, amount, source, jackpot_balance_after)
  VALUES (p_user_id, 'contribution', p_amount, p_source, v_new_total);

  RETURN v_new_total;
END;

== record_admin_revenue

DECLARE
  v_revenue_id uuid;
  v_currency   text;
BEGIN
  IF p_amount <= 0 THEN
    RETURN NULL;
  END IF;

  v_currency := UPPER(COALESCE(p_currency, 'FXU'));
  IF v_currency NOT IN ('FXU', 'USDT') THEN
    RAISE EXCEPTION 'Invalid currency for admin_revenue: %', v_currency;
  END IF;

  INSERT INTO public.admin_revenue (revenue_type, amount, currency, description, related_user_id, related_tx_id)
  VALUES (p_revenue_type, p_amount, v_currency, p_description, p_related_user_id, p_related_tx_id)
  RETURNING id INTO v_revenue_id;

  -- Reparto a accionistas (no-op si no hay accionistas activos)
  PERFORM public._distribute_shareholder_share(v_revenue_id, p_revenue_type, p_amount, v_currency, p_related_user_id);

  RETURN v_revenue_id;
END;

== process_completed_building_queue

DECLARE
  v_planet planets%ROWTYPE;
  v_queue_item RECORD;
  v_completed_count integer := 0;
  v_result jsonb;
  v_legacy_completed integer := 0;
BEGIN
  SELECT * INTO v_planet FROM planets WHERE id = p_planet_id;
  IF NOT FOUND OR v_planet.user_id != auth.uid() THEN
    RETURN jsonb_build_object('success', false, 'error', 'Planet not found or not owned');
  END IF;

  FOR v_queue_item IN
    SELECT bq.* FROM building_queue bq
    WHERE bq.planet_id = p_planet_id
    AND bq.end_time IS NOT NULL
    AND bq.end_time <= (NOW() + INTERVAL '2 seconds')
    ORDER BY bq.queue_position
    FOR UPDATE OF bq SKIP LOCKED
  LOOP
    v_result := complete_building_queue_item(v_queue_item.id);
    IF (v_result->>'success')::boolean THEN
      v_completed_count := v_completed_count + 1;
    END IF;
  END LOOP;

  SELECT COUNT(*) INTO v_legacy_completed
  FROM planet_buildings
  WHERE planet_id = p_planet_id
  AND is_upgrading = true
  AND upgrade_end_time IS NOT NULL
  AND upgrade_end_time <= (NOW() + INTERVAL '2 seconds');

  IF v_legacy_completed > 0 THEN
    UPDATE planets
    SET fields_used = fields_used + v_legacy_completed
    WHERE id = p_planet_id;

    UPDATE planet_buildings
    SET level = level + 1, is_upgrading = false, upgrade_end_time = NULL
    WHERE planet_id = p_planet_id
    AND is_upgrading = true
    AND upgrade_end_time IS NOT NULL
    AND upgrade_end_time <= (NOW() + INTERVAL '2 seconds');
  END IF;

  RETURN jsonb_build_object('success', true, 'completed_count', v_completed_count + v_legacy_completed);
END;
